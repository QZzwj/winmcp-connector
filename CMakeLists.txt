cmake_minimum_required(VERSION 3.21)
project(QZdeskConnector VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network)
qt_standard_project_setup()
qt_add_executable(QZdeskConnector WIN32 main.cpp)
target_link_libraries(QZdeskConnector PRIVATE Qt6::Widgets Qt6::Network)
if(MSVC)
    target_compile_options(QZdeskConnector PRIVATE /utf-8)
endif()
install(TARGETS QZdeskConnector RUNTIME DESTINATION .)
qt_generate_deploy_app_script(TARGET QZdeskConnector OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})
